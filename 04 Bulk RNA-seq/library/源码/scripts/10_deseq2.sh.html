#!/usr/bin/env bash
# 文件用途｜使用指定 Trinity/DESeq2 路线按比较表拟合差异。
# 运行方式：在项目根目录激活相应环境，先 source 本文件，再按同编号 MD 的顺序传入位置参数。source 本身
#   仅定义函数，不启动分析。
# 括号函数体在子 Shell 中运行；set -e 对未处理失败停止、-u 拒绝未定义变量、-o pipefail 使管道前段失败
#   可见。它不是自动恢复机制。
# set -a / source project.env / set +a（若本段使用）：临时把配置导出给子进程；不是创建 Conda 环境。双
#   引号保留文件名边界，$() 读取命令输出。
# mkdir -p：父目录不存在时一起创建，已有目录不清空；test -s/-e：检查非空文件/路径存在，! 取反。失败返
#   回非零，不自动下载缺失文件。
# 管道 | 传标准输出，> 写日志/文件时会覆盖同名目标，2>&1 把错误并入同一日志；/usr/bin/time -v -o 记录
#   资源详情（若本段调用）。不要把所有步骤都当作自动防覆盖。
# 本文件接口与输入输出：
# 输入：COUNT_MATRIX 的未归一化 counts、设计 samples.txt/contrasts.txt；输出 DE_analysis/repsN_cpmM。
#   已有目标目录会停止，不混用另一输入的模型。
# --matrix：计数表；--method DESeq2：指定检验软件；--samples_file：组与样本对应；--contrasts：实验/参
#   照组对。
# --min_reps_min_cpm：按 数量,CPM 拼接过滤门槛；--output：新统计目录。不同于后续只筛 padj/logFC 的汇总。

# 接口｜run_deseq2：使用指定 Trinity/DESeq2 路线按比较表拟合差异。
# 位置参数按以下顺序传入；Bash 没有 R 那样的具名形参调用，不能调换顺序。
# $1 min_reps：正整数；表达过滤要求至少多少个样本达到 min_cpm 条件，不是指定新的生物学重复数。与
#   min_cpm 联合改变进入 DESeq2 的基因。
# $2 min_cpm：非负 CPM 门槛（每百万计数）；至少 min_reps 个样本达到该过滤条件。改变后需要重新拟合，不
#   只是调整最后一张图。
# 返回/写出：成功返回退出状态 0，检查或软件失败为非零。实际结果保存在上述目录；本接口不返回一张内存中
#   的表。
run_deseq2() (
  set -euo pipefail
  [[ $# -eq 2 ]] || { echo "参数数量错误；请按第 10 章调用示例运行。" >&2; return 2; }
  min_reps="${1}"
  min_cpm="${2}"
  # 功能：按比较表运行原指定的 Trinity/DESeq2 分析，不以 TPM 代替 counts。
  # min_reps/min_cpm：至少多少个样本的 CPM 大于多少；改变后需重新拟合受影响比较。
  set -euo pipefail
  # min_reps：表达过滤需达到 CPM 条件的最少样本数，不等于模型的组数。
  # min_cpm：与 min_reps 配合的 CPM 下限；改变进入检验的基因集合，需要重做统计。
  set -a
  source 0script/project.env
  set +a
  design_dir="${DE_DESIGN_DIR:-5DESeq2/design}"
  analysis_dir="5DESeq2/DE_analysis/reps${min_reps}_cpm${min_cpm}"
  # 已有统计输出不能被新任务覆盖；更换模型/输入时明确另设 analysis_dir。
  if [[ -e "$analysis_dir" ]]; then
    echo "拒绝覆盖已有分析目录：$analysis_dir" >&2
    exit 1
  fi
  # Trinity 只创建最后一级；空项目需先准备父目录，不覆盖已有分析。
  mkdir -p 5DESeq2/DE_analysis
  run_DE_analysis.pl \
    --matrix "${COUNT_MATRIX:-3Salmon/quanti/gene.counts.matrix}" \
    --method DESeq2 \
    --samples_file "$design_dir/samples.txt" \
    --contrasts "$design_dir/contrasts.txt" \
    --min_reps_min_cpm "$min_reps,$min_cpm" \
    --output "$analysis_dir"
)
