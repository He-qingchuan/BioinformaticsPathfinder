#!/usr/bin/env bash
# 文件用途｜对已产生的 clean 双端 FASTQ 做独立质量检查。
# 运行方式：在项目根目录激活相应环境，先 source 本文件，再按同编号 MD 的顺序传入位置参数。source 本身
#   仅定义函数，不启动分析。
# 括号函数体在子 Shell 中运行；set -e 对未处理失败停止、-u 拒绝未定义变量、-o pipefail 使管道前段失败
#   可见。它不是自动恢复机制。
# set -a / source project.env / set +a（若本段使用）：临时把配置导出给子进程；不是创建 Conda 环境。双
#   引号保留文件名边界，$() 读取命令输出。
# mkdir -p：父目录不存在时一起创建，已有目录不清空；test -s/-e：检查非空文件/路径存在，! 取反。失败返
#   回非零，不自动下载缺失文件。
# 管道 | 传标准输出，> 写日志/文件时会覆盖同名目标，2>&1 把错误并入同一日志；/usr/bin/time -v -o 记录
#   资源详情（若本段调用）。不要把所有步骤都当作自动防覆盖。
# 本文件接口与输入输出：
# 输入：同一张样本表及 fastp/{sample_id}_clean_1/2.fastq.gz，不重新裁剪；输出 cleandata/qc 与
#   cleandata/multiqc。
# fastqc --threads：文件并行预算；--outdir：报告目录。multiqc --outdir/--filename：汇总目录/名称，--
#   force 更新同名 HTML。

# 接口｜run_clean_qc：对已产生的 clean 双端 FASTQ 做独立质量检查。
# 位置参数按以下顺序传入；Bash 没有 R 那样的具名形参调用，不能调换顺序。
# 参数：不接收位置参数；输入路径、样本和适用线程从当前项目读取，不在调用行硬写样本名。
# 返回/写出：成功返回退出状态 0，检查或软件失败为非零。实际结果保存在上述目录；本接口不返回一张内存中
#   的表。
run_clean_qc() (
  set -euo pipefail
  [[ $# -eq 0 ]] || { echo "参数数量错误；请按第 06 章调用示例运行。" >&2; return 2; }

  # 功能：对已经生成的 clean 双端 reads 做 FastQC/MultiQC，便于与原始报告比较。
  # 参数：THREADS 控制资源；输入按同一个样本表定位。
  set -euo pipefail
  set -a
  source 0script/project.env
  set +a
  mkdir -p 2data/cleandata/qc 2data/cleandata/multiqc
  clean_files=()
  sample_rows=$(Rscript 0script/tools/sample_columns.R sample_id)
  while IFS= read -r sample_id; do
    [[ $sample_id == sample_id ]] && continue
    clean_files+=("2data/cleandata/fastp/${sample_id}_clean_1.fastq.gz")
    clean_files+=("2data/cleandata/fastp/${sample_id}_clean_2.fastq.gz")
  done <<< "$sample_rows"
  [[ ${#clean_files[@]} -gt 0 ]]
  /usr/bin/time -v -o 2data/cleandata/qc/resources.log \
    fastqc --threads "$THREADS" --outdir 2data/cleandata/qc "${clean_files[@]}"
  multiqc 2data/cleandata/qc --outdir 2data/cleandata/multiqc --filename clean_multiqc.html --force
)
